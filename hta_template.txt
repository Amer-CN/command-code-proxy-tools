<script language="JScript">
var shell = new ActiveXObject("WScript.Shell");
var fso = new ActiveXObject("Scripting.FileSystemObject");

// 解析本文件所在目录（file:///F:/AIXM/command/xxx.hta -> F:\AIXM\command）
var loc = decodeURIComponent(String(window.location)).replace(/^file:\/\/\//, "").replace(/\//g, "\\");
var here = fso.GetParentFolderName(loc);
var binDir = here + "\\bin";
var exePath = binDir + "\\command-code-proxy.exe";
var keyFile = here + "\\api-key.txt";
var noticeFile = here + "\\notice_dismissed.flag";

var HEALTH = "http://127.0.0.1:55990/health";
var MODELS = "http://127.0.0.1:55990/v1/models";

__BASE64__

function httpGet(url) {
  try {
    var x = new ActiveXObject("MSXML2.XMLHTTP");
    x.open("GET", url, false);
    x.send();
    return (x.status == 200) ? x.responseText : null;
  } catch (e) { return null; }
}

function isRunning() { return httpGet(HEALTH) != null; }

function log(msg) {
  var el = document.getElementById("log");
  el.innerHTML = new Date().toLocaleTimeString() + "  " + msg + "<br>" + el.innerHTML;
}

function copyText(id) {
  var v = document.getElementById(id).value;
  window.clipboardData.setData("Text", v);
  log("已复制: " + v);
}

function copyModel(name) {
  window.clipboardData.setData("Text", name);
  log("已复制模型名: " + name);
}

function toggleKey() {
  var k = document.getElementById("apiKey");
  k.type = (k.type == "password") ? "text" : "password";
}

function toggleHelp() {
  var h = document.getElementById("help");
  h.style.display = (h.style.display == "block") ? "none" : "block";
}

function dismissNotice() {
  try {
    var f = fso.CreateTextFile(noticeFile, true);
    f.Close();
  } catch (e) {}
  document.getElementById("avNotice").style.display = "none";
  log("已关闭安全软件提示（下次不再显示）");
}

function refresh() {
  var running = isRunning();
  var dot = document.getElementById("dot");
  var st = document.getElementById("statusText");
  var btn = document.getElementById("toggleBtn");
  if (btn) {
    if (running) {
      btn.className = "danger";
      btn.innerHTML = "&#9632; 停止代理";
    } else {
      btn.className = "primary";
      btn.innerHTML = "&#9654; 启动代理";
    }
  }
  if (running) {
    dot.className = "dot green";
    st.innerHTML = "运行中 · http://127.0.0.1:55990 可用";
    loadModels();
  } else {
    dot.className = "dot gray";
    st.innerHTML = "未运行 · 点击「启动代理」";
    document.getElementById("modelList").innerHTML = "代理未运行时无法获取";
    document.getElementById("modelCount").innerHTML = "";
  }
}

function loadModels() {
  var txt = httpGet(MODELS);
  if (!txt) return;
  var list, i, html = "";
  try { list = eval("(" + txt + ")").data; } catch (e) { return; }
  for (i = 0; i < list.length; i++) {
    html += '<span class="chip" onclick="copyModel(\'' + list[i].id + '\')">' + list[i].id + "</span>";
  }
  document.getElementById("modelList").innerHTML = html;
  document.getElementById("modelCount").innerHTML = "（" + list.length + " 个）";
}

// ---- 内置组件释放 ----
function exeExists() { return fso.FileExists(exePath); }

function decodeExe() {
  try {
    if (!fso.FolderExists(binDir)) { fso.CreateFolder(binDir); }
    var dom = new ActiveXObject("MSXML2.DOMDocument.6.0");
    var node = dom.createElement("b64");
    node.dataType = "bin.base64";
    node.text = B64;
    var bytes = node.nodeTypedValue;
    var st = new ActiveXObject("ADODB.Stream");
    st.Type = 1;           // adTypeBinary
    st.Open();
    st.Write(bytes);
    st.SaveToFile(exePath, 2);  // adSaveCreateOverWrite
    st.Close();
    return true;
  } catch (e) {
    log("组件解压失败: " + e.message);
    return false;
  }
}

// ---- 启停 ----
function startProxy() {
  if (isRunning()) { log("代理已经在运行"); refresh(); return; }
  if (!exeExists()) {
    log("正在释放内置组件（首次约 10~60 秒）…");
    if (!decodeExe()) { refresh(); return; }
    log("组件释放完成");
  }
  var key = document.getElementById("apiKey").value;
  var cmd = '"' + exePath + '" -port 55990';
  if (key != "") {
    cmd += ' -api-key "' + key.replace(/"/g, '""') + '"';
    try {
      var f = fso.CreateTextFile(keyFile, true);
      f.Write(key);
      f.Close();
      log("API Key 已保存到 api-key.txt");
    } catch (e) {}
  }
  try {
    shell.Run(cmd, 0, false);  // 0 = 隐藏窗口后台运行
    log("已启动（后台运行），状态即将自动更新…");
  } catch (e) {
    log("启动失败: " + e.message);
  }
  setTimeout("refresh()", 1500);
}

function stopProxy() {
  try {
    shell.Run("taskkill /F /IM command-code-proxy.exe", 0, true);
    log("已发送停止命令");
  } catch (e) {
    log("停止失败: " + e.message);
  }
  setTimeout("refresh()", 1000);
}

function toggleProxy() {
  if (isRunning()) { stopProxy(); } else { startProxy(); }
}

// ---- 开机自启 ----
function setAutostart() {
  try {
    var startup = shell.SpecialFolders("Startup");
    var vbs = startup + "\\command-code-proxy-autostart.vbs";
    if (!exeExists()) {
      log("组件尚未释放，请先点「启动代理」");
      return;
    }
    var f = fso.CreateTextFile(vbs, true, false);
    f.WriteLine("Set sh = CreateObject(\"WScript.Shell\")");
    f.WriteLine("sh.Run \"" + exePath + " -port 55990\", 0, False");
    f.Close();
    log("开机自启已设置");
  } catch (e) {
    log("设置开机自启失败: " + e.message);
  }
}

function unsetAutostart() {
  try {
    var vbs = shell.SpecialFolders("Startup") + "\\command-code-proxy-autostart.vbs";
    if (fso.FileExists(vbs)) { fso.DeleteFile(vbs, true); }
    log("开机自启已取消");
  } catch (e) {
    log("取消失败: " + e.message);
  }
}

// ---- 初始化：自动解压 + 自动启动 ----
(function init() {
  window.resizeTo(560, 780);  // 初始窗口大小（可拖拽调整/最大化）
  try {
    if (fso.FileExists(keyFile)) {
      var f = fso.OpenTextFile(keyFile, 1);
      document.getElementById("apiKey").value = f.ReadAll();
      f.Close();
    }
  } catch (e) {}
  try {
    if (!fso.FileExists(noticeFile)) {
      document.getElementById("avNotice").style.display = "block";
    }
  } catch (e) {}
  if (exeExists()) {
    log("组件就绪");
    setTimeout("autoStart()", 200);
  } else {
    log("首次使用：正在释放内置组件，约 10~60 秒…");
    setTimeout("doFirstRun()", 100);
  }
  refresh();
  setInterval("refresh()", 3000);
})();

function doFirstRun() {
  if (decodeExe()) {
    log("组件释放完成");
    autoStart();
  } else {
    log("组件释放失败：请检查磁盘空间或杀毒软件设置，然后关闭本窗口重试");
    refresh();
  }
}

function autoStart() {
  if (!isRunning()) {
    log("自动启动代理…");
    startProxy();
  } else {
    log("代理已在运行");
    refresh();
  }
}
